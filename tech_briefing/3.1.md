<script>
(() => {
  'use strict';

  /* ---------- Theme ---------- */
  const root = document.documentElement;
  const themeBtn = document.getElementById('theme-toggle');
  const sunIcon = document.querySelector('.theme-icon-sun');
  const moonIcon = document.querySelector('.theme-icon-moon');

  const openSwitcherBtn = document.getElementById('open-report-switcher');
  if (openSwitcherBtn) {
    openSwitcherBtn.addEventListener('click', () => {
      if (typeof window.__openReportSwitcher === 'function') window.__openReportSwitcher();
    });
  }

  function applyTheme(theme) {
    root.setAttribute('data-theme', theme);
    if (theme === 'dark') {
      sunIcon.style.display = 'none';
      moonIcon.style.display = '';
      themeBtn.setAttribute('aria-label', 'Switch to light mode');
    } else {
      sunIcon.style.display = '';
      moonIcon.style.display = 'none';
      themeBtn.setAttribute('aria-label', 'Switch to dark mode');
    }
  }

  function getInitialTheme() {
    try {
      const stored = localStorage.getItem('report-theme');
      if (stored === 'light' || stored === 'dark') return stored;
    } catch (_) { /* localStorage may be blocked */ }
    return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
  }

  applyTheme(getInitialTheme());

  themeBtn.addEventListener('click', () => {
    const next = root.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
    applyTheme(next);
    try { localStorage.setItem('report-theme', next); } catch (_) {}
  });

  /* ---------- Reading time ---------- */
  const article = document.querySelector('.article');
  if (article) {
    const text = article.innerText || article.textContent || '';
    const words = text.trim().split(/\s+/).length;
    const minutes = Math.max(1, Math.round(words / 225));
    const rt = document.getElementById('reading-time');
    if (rt) rt.textContent = `~${minutes} min read`;
  }

  /* ---------- ToC scroll-spy ---------- */
  const tocLinks = Array.from(document.querySelectorAll('.toc__link'));
  const tocMap = new Map();
  tocLinks.forEach(link => {
    const id = link.getAttribute('href').slice(1);
    const target = document.getElementById(id);
    if (target) tocMap.set(target, link);
  });

  const indicatorLabel = document.querySelector('.section-indicator__label');

  if ('IntersectionObserver' in window && tocMap.size > 0) {
    let activeLink = null;
    const observer = new IntersectionObserver((entries) => {
      // Find the topmost intersecting heading
      const visible = entries
        .filter(e => e.isIntersecting)
        .sort((a, b) => a.boundingClientRect.top - b.boundingClientRect.top);
      if (visible.length > 0) {
        const link = tocMap.get(visible[0].target);
        if (link && link !== activeLink) {
          if (activeLink) activeLink.classList.remove('is-active');
          link.classList.add('is-active');
          activeLink = link;
          if (indicatorLabel) indicatorLabel.textContent = link.textContent.trim();
        }
      }
    }, {
      rootMargin: '-10% 0px -70% 0px',
      threshold: [0, 1]
    });

    tocMap.forEach((_link, target) => observer.observe(target));
  }

  /* ---------- Sidenote fade-in (phase 2) ---------- */
  try {
    const sidenotes = document.querySelectorAll('.sidenote');
    const reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;

    if (reduced || !('IntersectionObserver' in window) || sidenotes.length === 0) {
      sidenotes.forEach(sn => sn.classList.add('is-revealed'));
    } else {
      const sidenoteObserver = new IntersectionObserver(entries => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.querySelectorAll('.sidenote').forEach(sn =>
              sn.classList.add('is-revealed')
            );
            sidenoteObserver.unobserve(entry.target);
          }
        });
      }, { rootMargin: '0px 0px -30% 0px' });

      sidenotes.forEach(sn => {
        const para = sn.closest('p, li, blockquote') || sn.parentElement;
        if (para) sidenoteObserver.observe(para);
      });
    }
  } catch (_) {
    document.querySelectorAll('.sidenote').forEach(sn => sn.classList.add('is-revealed'));
  }

  /* ---------- Section indicator h1-visibility observer (phase 3) ---------- */
  const indicator = document.getElementById('section-indicator');
  const h1 = document.querySelector('article h1');
  const hasH2 = document.querySelector('article h2') !== null;

  if (indicator && h1 && hasH2 && 'IntersectionObserver' in window) {
    const h1Observer = new IntersectionObserver(entries => {
      const h1Visible = entries.some(e => e.isIntersecting);
      indicator.classList.toggle(
        'is-visible',
        !h1Visible && indicatorLabel && indicatorLabel.textContent.length > 0
      );
    }, { threshold: 0 });
    h1Observer.observe(h1);
  }

  /* ---------- ToC mobile accordion ---------- */
  const tocRail = document.getElementById('toc-rail');
  const tocToggle = tocRail.querySelector('.toc-rail__toggle');
  if (tocToggle) {
    tocToggle.addEventListener('click', () => {
      const open = tocRail.getAttribute('data-open') === 'true';
      tocRail.setAttribute('data-open', String(!open));
      tocToggle.setAttribute('aria-expanded', String(!open));
    });
    // Close after navigating on mobile
    tocLinks.forEach(link => link.addEventListener('click', () => {
      if (window.matchMedia('(max-width: 1023px)').matches) {
        tocRail.setAttribute('data-open', 'false');
        tocToggle.setAttribute('aria-expanded', 'false');
      }
    }));
  }

  /* ---------- Copy buttons on code blocks ---------- */
  document.querySelectorAll('.codeblock').forEach(block => {
    const btn = block.querySelector('.codeblock__copy');
    const code = block.querySelector('pre code');
    if (!btn || !code) return;

    btn.addEventListener('click', async () => {
      const text = code.innerText;
      try {
        await navigator.clipboard.writeText(text);
      } catch (_) {
        // Fallback
        const ta = document.createElement('textarea');
        ta.value = text;
        document.body.appendChild(ta);
        ta.select();
        try { document.execCommand('copy'); } catch (_) {}
        document.body.removeChild(ta);
      }
      const label = btn.querySelector('.copy-label');
      const originalText = label ? label.textContent : 'Copy';
      btn.classList.add('is-copied');
      if (label) label.textContent = 'Copied';
      setTimeout(() => {
        btn.classList.remove('is-copied');
        if (label) label.textContent = originalText;
      }, 1400);
    });
  });

  /* ---------- Reading progress bar ---------- */
  const progressBar = document.getElementById('progress-bar');
  function updateProgress() {
    const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    const scrolled = window.scrollY;
    const pct = docHeight > 0 ? Math.min(100, Math.max(0, (scrolled / docHeight) * 100)) : 0;
    progressBar.style.width = pct + '%';
  }
  let progressFrame = null;
  window.addEventListener('scroll', () => {
    if (progressFrame) return;
    progressFrame = requestAnimationFrame(() => {
      updateProgress();
      progressFrame = null;
    });
  }, { passive: true });
  updateProgress();

  /* ---------- Hover footnote popovers ---------- */
  const fnrefs = Array.from(document.querySelectorAll('.fnref'));
  let popover = null;
  let hoverTimer = null;

  function buildPopover() {
    if (popover) return popover;
    popover = document.createElement('div');
    popover.className = 'fnref-popover';
    popover.setAttribute('data-visible', 'false');
    document.body.appendChild(popover);
    return popover;
  }

  function showPopover(ref) {
    if (window.matchMedia('(hover: none)').matches) return;
    const id = ref.getAttribute('href').slice(1);
    const fn = document.getElementById(id);
    if (!fn) return;
    const p = buildPopover();
    // Strip the back-link
    const html = Array.from(fn.cloneNode(true).childNodes)
      .filter(n => !(n.nodeType === 1 && n.classList && n.classList.contains('fn-back')))
      .map(n => n.outerHTML || n.textContent)
      .join('');
    p.innerHTML = html;
    p.setAttribute('data-visible', 'true');
    const rect = ref.getBoundingClientRect();
    const popRect = p.getBoundingClientRect();
    let left = rect.left + window.scrollX;
    let top = rect.top + window.scrollY - popRect.height - 8;
    // Clamp horizontally
    const maxLeft = window.scrollX + document.documentElement.clientWidth - popRect.width - 12;
    if (left > maxLeft) left = maxLeft;
    if (left < 12) left = 12;
    // Flip below if it would go off the top
    if (top < window.scrollY + 8) top = rect.bottom + window.scrollY + 8;
    p.style.left = left + 'px';
    p.style.top = top + 'px';
  }

  function hidePopover() {
    if (popover) popover.setAttribute('data-visible', 'false');
  }

  fnrefs.forEach(ref => {
    ref.addEventListener('mouseenter', () => {
      clearTimeout(hoverTimer);
      hoverTimer = setTimeout(() => showPopover(ref), 200);
    });
    ref.addEventListener('mouseleave', () => {
      clearTimeout(hoverTimer);
      setTimeout(hidePopover, 120);
    });
    ref.addEventListener('focus', () => showPopover(ref));
    ref.addEventListener('blur', hidePopover);
  });
  document.addEventListener('scroll', hidePopover, { passive: true });

  /* ---------- Edit mode ---------- */
  const editBtn = document.getElementById('edit-toggle');
  let wasEdited = false;
  let editEverEntered = false;

  function setEditMode(on) {
    document.body.setAttribute('data-edit-mode', String(!!on));
    if (article) article.contentEditable = on ? 'true' : 'false';
    if (editBtn) {
      editBtn.setAttribute('aria-pressed', String(!!on));
      editBtn.classList.toggle('is-active', !!on);
    }
    if (on) {
      editEverEntered = true;
      // Move caret into the article so typing works immediately.
      try {
        const firstP = article.querySelector('p');
        if (firstP) {
          const sel = window.getSelection();
          const range = document.createRange();
          range.selectNodeContents(firstP);
          range.collapse(false);
          sel.removeAllRanges();
          sel.addRange(range);
        }
      } catch (_) {}
    } else {
      // Drop selection inside article to avoid lingering caret.
      try { window.getSelection().removeAllRanges(); } catch (_) {}
    }
  }

  if (editBtn && article) {
    editBtn.addEventListener('click', () => {
      const on = document.body.getAttribute('data-edit-mode') !== 'true';
      setEditMode(on);
    });
    article.addEventListener('input', () => { wasEdited = true; }, { passive: true });
  }

  /* ---------- Save as Markdown ---------- */
  function saveMarkdown() {
    const sourceEl = document.getElementById('source-md');
    const titleEl = article.querySelector('h1');
    const title = titleEl ? titleEl.textContent.trim() : 'report';
    const slug = title.toLowerCase()
      .replace(/['"]/g, '')
      .replace(/[^a-z0-9]+/g, '-')
      .replace(/^-+|-+$/g, '')
      .slice(0, 60);
    const date = new Date().toISOString().slice(0, 10);
    const filename = `${slug || 'report'}-${date}.md`;

    // Prefer the live DOM whenever the user has edited; fall back to embedded source otherwise.
    let md;
    if ((wasEdited || editEverEntered) && article) {
      md = domToMarkdown(article);
    } else if (sourceEl && sourceEl.textContent.trim()) {
      md = sourceEl.textContent.replace(/^\n/, '');
    } else {
      md = domToMarkdown(article);
    }

    // In review mode, write the edit back to the source file so any agent
    // reading the .md sees the change — same channel as comments. Outside
    // review mode (file:// open), fall back to a blob download.
    if (document.documentElement.dataset.reviewMode === 'true') {
      fetch('/api/source', {
        method: 'PUT',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ markdown: md }),
      }).then(async res => {
        if (!res.ok) throw new Error((await res.text()) || `HTTP ${res.status}`);
        return res.json();
      }).then(payload => {
        // Reflect what was saved so a follow-up save isn't computed against stale
        // embedded source. The server may have re-injected preserved comments.
        if (sourceEl) sourceEl.textContent = '\n' + md;
        wasEdited = false;
        editEverEntered = false;
        const saved = payload && payload.file ? payload.file : filename;
        showToast(`Saved to <code>${escapeHtml(saved)}</code>`);
      }).catch(err => {
        showToast(`Save failed: ${escapeHtml(err.message || String(err))}`);
      });
      return;
    }

    const blob = new Blob([md], { type: 'text/markdown;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 1000);

    showToast(`Saved as <code>${escapeHtml(filename)}</code>`);
  }

  function escapeHtml(s) {
    return s.replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
  }

  /* ---------- DOM → Markdown serializer (Tier 2) ----------
     Round-trips: frontmatter, lede, callouts (NOTE/INFO/WARNING/INSIGHT/TIP/DANGER),
     definition lists, tables with right-aligned numeric columns, footnote refs and
     definitions, sidenotes, figure-as-chart-anchor, code blocks with language and path.
  */
  function domToMarkdown(rootEl) {
    const headerEl = rootEl.querySelector('header.article-header');
    // Footnote defs are rendered inline at the position of section.footnotes
    // (handled by walkBlock); only the article header is excluded from the body walk.
    const skip = new Set();
    if (headerEl) skip.add(headerEl);

    let out = '';

    // Frontmatter from header.article-header dl.meta + h1
    const fm = collectFrontmatter(headerEl);
    out += frontmatterToYaml(fm);

    // Title
    const h1 = headerEl ? headerEl.querySelector('h1') : null;
    if (h1) out += `# ${stripAnchors(h1).trim()}\n\n`;

    // Lede (rendered as a top-level blockquote, matching the source convention)
    const ledeP = headerEl ? headerEl.querySelector('p.lede') : null;
    if (ledeP) out += `> ${inlineWalk(ledeP).trim()}\n\n`;

    // Body
    for (const child of rootEl.children) {
      if (skip.has(child)) continue;
      out += walkBlock(child, skip);
    }

    return out.replace(/\n{3,}/g, '\n\n').trim() + '\n';
  }

  function stripAnchors(el) {
    const clone = el.cloneNode(true);
    clone.querySelectorAll('a.heading-anchor, a.fn-back').forEach(n => n.remove());
    return clone.textContent;
  }

  function collectFrontmatter(headerEl) {
    if (!headerEl) return null;
    const meta = headerEl.querySelector('dl.meta');
    const h1 = headerEl.querySelector('h1');
    const fm = {};
    if (h1) fm.title = h1.textContent.trim();
    if (!meta) return fm;
    let key = null;
    for (const child of meta.children) {
      const tag = child.tagName;
      if (tag === 'DT') {
        key = child.textContent.trim().toLowerCase().replace(/\s+/g, '_');
      } else if (tag === 'DD' && key) {
        const tags = child.querySelectorAll('.tag');
        if (tags.length) {
          fm[key] = Array.from(tags).map(t => t.textContent.trim());
        } else {
          let v = child.textContent.trim();
          // The HTML reading-time often reads "~12 min read"; keep the friendly form trimmed.
          if (key === 'reading_time') v = v.replace(/^~/, '').replace(/\s+read$/, '');
          fm[key] = v;
        }
        key = null;
      }
    }
    return fm;
  }

  function frontmatterToYaml(fm) {
    if (!fm || !Object.keys(fm).length) return '';
    const order = ['title', 'generated_by', 'date', 'reading_time', 'tags', 'sources', 'version'];
    const seen = new Set();
    const lines = ['---'];
    function emit(k, v) {
      if (Array.isArray(v)) lines.push(`${k}: [${v.join(', ')}]`);
      else lines.push(`${k}: ${v}`);
    }
    for (const k of order) {
      if (k in fm) { emit(k, fm[k]); seen.add(k); }
    }
    for (const [k, v] of Object.entries(fm)) {
      if (!seen.has(k)) emit(k, v);
    }
    lines.push('---');
    return lines.join('\n') + '\n\n';
  }

  function collectFootnoteDefs(section) {
    if (!section) return '';
    const items = Array.from(section.querySelectorAll('ol > li[id^="fn"]'));
    if (!items.length) return '';
    return items.map(li => {
      const num = li.id.replace(/^fn/, '');
      const clone = li.cloneNode(true);
      clone.querySelectorAll('.fn-back').forEach(n => n.remove());
      return `[^${num}]: ${inlineWalk(clone).trim()}`;
    }).join('\n\n');
  }

  /* ---- Inline serializer: produces single-line markdown for cell/footnote contexts ---- */
  function inlineWalk(el) {
    let out = '';
    el.childNodes.forEach(n => {
      if (n.nodeType === 3) { out += n.textContent; return; }
      if (n.nodeType !== 1) return;
      const tag = n.tagName.toLowerCase();
      if (n.classList && (n.classList.contains('heading-anchor') || n.classList.contains('fn-back'))) return;
      if (n.classList && n.classList.contains('fnref')) {
        const fn = n.dataset.fn || (n.textContent.match(/\d+/) || [''])[0];
        // Glue the ref to the preceding word: strip a single trailing space
        // (and a trailing nbsp), matching the standard `flows.[^1]` convention.
        out = out.replace(/[  ]$/, '');
        out += `[^${fn}]`;
        return;
      }
      const inner = inlineWalk(n);
      switch (tag) {
        case 'em': case 'i':       out += `*${inner}*`; break;
        case 'strong': case 'b':   out += `**${inner}**`; break;
        case 'code':               out += `\`${inner}\``; break;
        case 'mark':               out += `==${inner}==`; break;
        case 'br':                 out += ' '; break;
        case 'a': {
          const href = n.getAttribute('href') || '';
          out += `[${inner}](${href})`;
          break;
        }
        case 'small': case 'span': case 'dfn': out += inner; break;
        case 'p': case 'div':      out += inner; break;
        default:                   out += inner;
      }
    });
    return out.replace(/\s+/g, ' ').trim();
  }

  /* ---- Block serializer: handles the full document tree ---- */
  function walkBlock(node, skip) {
    if (node.nodeType === 3) return node.textContent;
    if (node.nodeType !== 1) return '';
    const el = node;
    if (skip && skip.has(el)) return '';

    // Hidden/decorative
    if (el.matches('a.heading-anchor, a.fn-back, .codeblock__copy, .codeblock__header, .sidenote-rail, .end-actions')) return '';

    // Review-mode overlays are painted into the article DOM but are not part
    // of the source markdown — exclude them so an in-browser save does not
    // leak comment text back into the .md.
    if (el.matches('.report-comment-margin')) return '';

    // Footnote definitions render inline at the section's document position,
    // headed by their `[^N]:` markers — not by the section's "Footnotes" h2.
    if (el.matches('section.footnotes')) {
      const defs = collectFootnoteDefs(el);
      return defs ? defs + '\n\n' : '';
    }

    // Class-based dispatch (most specific first)
    if (el.matches('blockquote.callout') || el.matches('div.callout-box')) {
      return renderCallout(el);
    }
    if (el.matches('aside.sidenote')) {
      return renderSidenote(el);
    }
    if (el.matches('div.codeblock')) {
      return renderCodeblock(el);
    }
    if (el.matches('figure.framed')) {
      return renderFigure(el);
    }
    if (el.matches('div.table-wrap')) {
      return Array.from(el.children).map(c => walkBlock(c, skip)).join('');
    }
    if (el.matches('table')) {
      return renderTable(el);
    }
    if (el.matches('dl.meta')) {
      return ''; // emitted as frontmatter
    }
    if (el.matches('dl')) {
      return renderDl(el);
    }
    if (el.matches('p.eyebrow')) {
      return ''; // not represented in source markdown
    }
    if (el.matches('p.section-lede')) {
      // The HTML wraps the lede in <em>; walk into it so we don't double-wrap
      // and produce **bold** when the source was *italic*.
      const em = el.querySelector(':scope > em');
      const text = em ? inlineWalk(em).trim() : inlineWalk(el).trim();
      return `*${text}*\n\n`;
    }
    if (el.matches('p.dropcap')) {
      return `${inlineWalk(el).trim()}\n\n`;
    }
    if (el.matches('p.pullquote')) {
      return `> ${inlineWalk(el).trim()}\n\n`;
    }
    if (el.matches('p.lede')) {
      return `> ${inlineWalk(el).trim()}\n\n`;
    }
    if (el.matches('section, main, article, header, .end-actions')) {
      return Array.from(el.children).map(c => walkBlock(c, skip)).join('');
    }

    const tag = el.tagName.toLowerCase();
    switch (tag) {
      case 'h1': return `\n# ${inlineWalk(el).trim()}\n\n`;
      case 'h2': return `\n## ${inlineWalk(el).trim()}\n\n`;
      case 'h3': return `\n### ${inlineWalk(el).trim()}\n\n`;
      case 'h4': return `\n#### ${inlineWalk(el).trim()}\n\n`;
      case 'h5': return `\n##### ${inlineWalk(el).trim()}\n\n`;
      case 'h6': return `\n###### ${inlineWalk(el).trim()}\n\n`;
      case 'p':  return `${inlineWalk(el).trim()}\n\n`;
      case 'pre': return `\n\`\`\`\n${el.textContent.trim()}\n\`\`\`\n\n`;
      case 'ul': return Array.from(el.children).map(li => `- ${inlineWalk(li).trim()}`).join('\n') + '\n\n';
      case 'ol': return Array.from(el.children).map((li, i) => `${i+1}. ${inlineWalk(li).trim()}`).join('\n') + '\n\n';
      case 'blockquote': {
        const inner = Array.from(el.childNodes).map(n => walkBlock(n, skip)).join('').trim();
        return inner.split('\n').map(l => l.length ? `> ${l}` : '>').join('\n') + '\n\n';
      }
      case 'hr': return '\n---\n\n';
      case 'aside':
      case 'div':
      case 'span':
        return Array.from(el.childNodes).map(n => walkBlock(n, skip)).join('');
      case 'figure':
      case 'figcaption':
      case 'svg':
        return '';
      default:
        return Array.from(el.childNodes).map(n => walkBlock(n, skip)).join('');
    }
  }

  function renderCallout(el) {
    const type = (el.dataset.callout || 'note').toUpperCase();
    let bodyEl = el.querySelector('.callout-box__body');
    if (!bodyEl) {
      bodyEl = el.cloneNode(true);
      bodyEl.querySelectorAll('.callout__label, .callout-box__icon, svg').forEach(n => n.remove());
    }
    // Render body as block content, then prefix lines with `>`.
    const parts = [];
    for (const child of bodyEl.childNodes) {
      const piece = walkBlock(child, null);
      if (piece) parts.push(piece);
    }
    const inner = parts.join('').replace(/\n{3,}/g, '\n\n').trim();
    if (!inner) return `> [${type}]\n\n`;
    const lines = inner.split('\n');
    const out = lines.map((l, i) => {
      if (i === 0) return l.length ? `> [${type}] ${l}` : `> [${type}]`;
      return l.length ? `> ${l}` : '>';
    }).join('\n');
    return out + '\n\n';
  }

  function renderSidenote(el) {
    const numEl = el.querySelector('.sidenote__num');
    const num = numEl ? numEl.textContent.trim() : '';
    // If a matching footnote definition exists in the formal section, skip this — the Footnotes
    // section is the canonical source and will be appended at the end.
    if (num && document.getElementById(`fn${num}`)) return '';
    const clone = el.cloneNode(true);
    const cn = clone.querySelector('.sidenote__num');
    if (cn) cn.remove();
    const text = inlineWalk(clone).trim();
    if (!text) return '';
    return num ? `\n[^${num}]: ${text}\n\n` : `\n> ${text}\n\n`;
  }

  function renderCodeblock(el) {
    const lang = el.dataset.lang || el.getAttribute('data-lang') || '';
    const pathEl = el.querySelector('.codeblock__path');
    const codeEl = el.querySelector('pre code');
    if (!codeEl) return '';
    let code = codeEl.textContent.replace(/\s+$/, '');
    if (pathEl && pathEl.textContent.trim()) {
      const path = pathEl.textContent.trim();
      const firstLine = code.split('\n', 1)[0] || '';
      // Don't double-prepend if the path already appears in the first line.
      if (!firstLine.includes(path)) {
        code = `// ${path}\n${code}`;
      }
    }
    return `\n\`\`\`${lang}\n${code}\n\`\`\`\n\n`;
  }

  function renderFigure(el) {
    const cap = el.querySelector('figcaption');
    const caption = cap ? cap.textContent.trim() : '';
    // Prefer the SVG <title> as the slug source — it's the chart's semantic name,
    // typically shorter than the figcaption, and matches the convention used in
    // the source markdown anchors. Fall back to the figcaption if absent.
    const svgTitle = el.querySelector('svg > title, svg title');
    const slugSource = (svgTitle ? svgTitle.textContent : caption) || 'chart';
    const slug = slugSource.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 60);
    const target = slug ? `chart-${slug}` : 'chart';
    return `\n![${caption}](#${target})\n\n`;
  }

  function renderTable(el) {
    const headRow = el.querySelector('thead tr');
    if (!headRow) return '';
    const headers = Array.from(headRow.children);
    const headerLine = '| ' + headers.map(th => inlineWalk(th).trim()).join(' | ') + ' |';
    const sepLine = '|' + headers.map(th => th.classList.contains('num') ? '---:' : '---').join('|') + '|';
    const bodyRows = Array.from(el.querySelectorAll('tbody tr')).map(tr =>
      '| ' + Array.from(tr.children).map(td => inlineWalk(td).trim()).join(' | ') + ' |'
    );
    return [headerLine, sepLine, ...bodyRows].join('\n') + '\n\n';
  }

  function renderDl(el) {
    const items = Array.from(el.children);
    const blocks = [];
    let i = 0;
    while (i < items.length) {
      if (items[i].tagName === 'DT') {
        const term = inlineWalk(items[i]).trim();
        const defs = [];
        i++;
        while (i < items.length && items[i].tagName === 'DD') {
          defs.push(`: ${inlineWalk(items[i]).trim()}`);
          i++;
        }
        blocks.push(term + '\n' + defs.join('\n'));
      } else {
        i++;
      }
    }
    return blocks.join('\n\n') + '\n\n';
  }

  /* ---------- Server-backed review mode ---------- */
  function initReviewMode() {
    if (document.documentElement.dataset.reviewMode !== 'true') return;
    const commentToggle = document.getElementById('review-comment-toggle');
    const bannerClose = document.getElementById('review-mode-banner-close');
    const reviewRail = document.getElementById('review-rail');
    const reviewTray = document.getElementById('review-tray');
    let reviewComments = [];
    let activeFilter = 'open';
    let activeCommentId = '';
    let reviewTrayOpen = false;

    /* ---------- popup (unified comment writing surface) ---------- */
    const popup = (() => {
      let current = null;
      // current shape: { wrapper, atom, state, initialBody, originEl, mode, comment, targetBlockId }

      function open({ mode, target, comment, originEl }) {
        close({ checkDirty: false });
        const blockId = target || comment?.target || '';
        const targetBlock = blockId ? findBlock(blockId) : null;
        const state = { mode, blockId, comment };
        const atom = buildAtom(state);
        if (!atom) return;
        let wrapperKind = pickMode();
        let wrapper = wrapperKind === 'rail' ? mountRail(targetBlock) : null;
        if (!wrapper) { wrapperKind = 'float'; wrapper = mountFloat(); }
        wrapper.appendChild(atom);
        atom.classList.add(wrapperKind === 'float' ? 'report-comment-pop--float' : 'report-comment-pop--rail');
        if (wrapperKind === 'float') placeFloat(originEl || targetBlock, atom);
        const initialBody = (mode === 'edit' && comment?.body) ? comment.body : '';
        current = { wrapper, atom, state, initialBody, originEl, mode, comment, targetBlockId: blockId };
        bindKeys(atom);
        const ta = atom.querySelector('.report-comment-pop__field');
        if (mode === 'edit' && ta) ta.value = initialBody;
        requestAnimationFrame(() => ta?.focus());
      }

      function close({ checkDirty = true } = {}) {
        if (!current) return;
        if (checkDirty && isDirty()) return requestDiscardConfirm();
        const { wrapper, originEl } = current;
        const railParent = wrapper?.parentElement?.matches('.report-comment-margin[data-report-comment-ephemeral="true"]') ? wrapper.parentElement : null;
        wrapper?.remove();
        if (railParent && railParent.children.length === 0) railParent.remove();
        current = null;
        if (originEl && typeof originEl.focus === 'function') originEl.focus();
      }

      function isDirty() {
        if (!current) return false;
        const ta = current.atom?.querySelector('.report-comment-pop__field');
        return ta ? ta.value !== current.initialBody : false;
      }

      function pickMode() {
        if (typeof window === 'undefined' || !window.matchMedia) return 'float';
        return window.matchMedia('(min-width: 1280px)').matches ? 'rail' : 'float';
      }

      function mountRail(targetBlock) {
        // Reuse the existing right-margin column (.report-comment-margin
        // with data-report-comment-target="block:<id>"). report-comment-pop--rail
        // is applied by the caller; the wrapper is structural only — no chrome.
        const target = targetBlock?.dataset?.reportBlockId;
        if (!target) return null;
        const wrapper = document.createElement('div');
        wrapper.className = 'report-comment-pop--rail-wrapper';
        const selector = `.report-comment-margin[data-report-comment-target="${CSS.escape(target)}"]`;
        let margin = document.querySelector(selector);
        if (!margin) {
          margin = document.createElement('aside');
          margin.className = 'report-comment-margin';
          margin.setAttribute('aria-label', `Review comments for ${target}`);
          margin.dataset.reportCommentTarget = target;
          margin.dataset.reportCommentEphemeral = 'true';
          targetBlock.insertAdjacentElement('afterend', margin);
        }
        margin.insertBefore(wrapper, margin.firstChild);
        return wrapper;
      }

      function mountFloat(/* originRect */) {
        const wrapper = document.createElement('div');
        wrapper.className = 'report-comment-pop report-comment-pop--float-wrapper';
        document.body.appendChild(wrapper);
        return wrapper;
      }

      function placeFloat(anchor, popupEl) {
        if (!anchor || !popupEl) return;
        // Pre-open: if anchor is below the fold by more than 30% of the viewport,
        // smooth-scroll so it sits near the top.
        const anchorRect = anchor.getBoundingClientRect();
        const vh = window.innerHeight;
        if (anchorRect.top > vh * 0.3) window.scrollBy({ top: anchorRect.top - vh * 0.3, behavior: 'smooth' });
        const popupRect = popupEl.getBoundingClientRect();
        const vw = window.innerWidth;
        const scrollY = window.scrollY;
        const scrollX = window.scrollX;
        const gap = 8;
        let top = anchorRect.bottom + scrollY + gap;
        let left = anchorRect.left + scrollX;
        // Right-edge clip → shift left.
        if (left + popupRect.width > scrollX + window.innerWidth - gap) left = scrollX + window.innerWidth - popupRect.width - gap;
        if (left < scrollX + gap) left = scrollX + gap;
        // Bottom clip → flip above the anchor.
        const flipUp = anchorRect.bottom + popupRect.height + gap > vh;
        if (flipUp) top = anchorRect.top + scrollY - popupRect.height - gap;
        popupEl.style.left = `${left}px`;
        popupEl.style.top = `${top}px`;
        if (flipUp) popupEl.setAttribute('data-tail-side', 'above'); else popupEl.setAttribute('data-tail-side', 'below');
      }

      function buildAtom(state) {
        // state: { mode, blockId, comment }
        const headingId = `report-comment-pop-head-${Math.random().toString(36).slice(2, 8)}`;
        const isEdit = state.mode === 'edit';
        const statusLabel = isEdit
          ? `#${escapeText(state.comment?.id || '')}`
          : 'New comment';
        const pill = isEdit
          ? `<button type="button" class="report-comment-pop__pill" data-action="toggle-status">${state.comment?.status === 'resolved' ? 'resolved' : 'unresolved'}</button>`
          : '';
        const deleteBtn = isEdit
          ? `<button type="button" data-action="delete">Delete</button>`
          : '';

        const wrapper = document.createElement('div');
        wrapper.innerHTML = `
          <form class="report-comment-pop" role="dialog" aria-labelledby="${headingId}">
            <div class="report-comment-pop__head" id="${headingId}">
              <span><span class="report-comment-pop__id">${escapeText((state.blockId || state.comment?.target || '').replace(/^block:/, ''))}</span> · <span class="report-comment-pop__status">${statusLabel}</span></span>
              ${pill}
            </div>
            <div class="report-comment-pop__body">
              <textarea class="report-comment-pop__field" aria-label="Comment text">${escapeText(state.comment?.body || '')}</textarea>
              <p class="report-comment-pop__message" aria-live="polite"></p>
            </div>
            <div class="report-comment-pop__hint">
              <span class="report-comment-pop__keys"><kbd>⌘</kbd><kbd>↵</kbd> save · <kbd>Esc</kbd> close</span>
              <span class="report-comment-pop__actions">
                ${deleteBtn}
                <button type="button" data-action="cancel">Cancel</button>
                <button type="submit" data-action="save">Save</button>
              </span>
            </div>
          </form>
        `;
        return wrapper.firstElementChild;
      }

      function bindKeys(atomEl) {
        atomEl.querySelector('[data-action="cancel"]')?.addEventListener('click', () => close());
        atomEl.querySelector('[data-action="delete"]')?.addEventListener('click', (event) => { event.preventDefault(); deleteCurrent(); });
        atomEl.querySelector('[data-action="toggle-status"]')?.addEventListener('click', (event) => { event.preventDefault(); requestStatusToggleConfirm(); });
        atomEl.addEventListener('submit', (event) => { event.preventDefault(); save(); });
        atomEl.addEventListener('keydown', (event) => {
          const cmd = event.metaKey || event.ctrlKey;
          if (event.key === 'Enter' && cmd) { event.preventDefault(); save(); return; }
          if (event.key === 'Escape') { event.preventDefault(); close(); return; }
          if (event.key === 'Tab') trapFocus(event, atomEl);
        });
      }

      function trapFocus(event, atomEl) {
        const focusables = atomEl.querySelectorAll('textarea, button');
        if (focusables.length === 0) return;
        const first = focusables[0];
        const last = focusables[focusables.length - 1];
        if (event.shiftKey && document.activeElement === first) { event.preventDefault(); last.focus(); }
        else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
      }

      let deleteConfirmed = false;

      function requestDiscardConfirm() {
        if (!current) return;
        renderConfirmStrip({
          prompt: 'Discard changes?',
          confirmLabel: 'Discard',
          cancelLabel: 'Keep',
          onConfirm: () => close({ checkDirty: false }),
        });
      }

      function requestDeleteConfirm() {
        if (!current || current.mode !== 'edit') return;
        renderConfirmStrip({
          prompt: `Delete #${current.comment.id}?`,
          confirmLabel: 'Delete',
          cancelLabel: 'Cancel',
          danger: true,
          onConfirm: () => { deleteConfirmed = true; deleteCurrent(); },
        });
      }

      function requestStatusToggleConfirm() {
        if (!current || current.mode !== 'edit') return;
        const next = current.comment.status === 'resolved' ? 'open' : 'resolved';
        renderConfirmStrip({
          prompt: next === 'resolved' ? 'Mark resolved?' : 'Reopen?',
          confirmLabel: next === 'resolved' ? 'Resolve' : 'Reopen',
          cancelLabel: 'Cancel',
          onConfirm: () => doStatusToggle(next),
        });
      }

      function renderConfirmStrip({ prompt, confirmLabel, cancelLabel, danger, onConfirm }) {
        if (!current) return;
        const hint = current.atom.querySelector('.report-comment-pop__hint');
        if (!hint) return;
        const previous = hint.innerHTML;
        hint.innerHTML = `
          <span class="report-comment-pop__keys">${escapeText(prompt)}</span>
          <span class="report-comment-pop__actions">
            <button type="button" data-confirm="cancel">${escapeText(cancelLabel)}</button>
            <button type="button" data-confirm="ok" data-danger="${danger ? 'true' : 'false'}">${escapeText(confirmLabel)}</button>
          </span>
        `;
        const restore = () => { hint.innerHTML = previous; };
        hint.querySelector('[data-confirm="cancel"]').addEventListener('click', restore);
        hint.querySelector('[data-confirm="ok"]').addEventListener('click', () => { restore(); onConfirm(); });
      }

      async function doStatusToggle(nextStatus) {
        if (!current || current.mode !== 'edit') return;
        const message = current.atom.querySelector('.report-comment-pop__message');
        setActionsBusy(true, nextStatus === 'resolved' ? 'Resolving…' : 'Reopening…');
        try {
          const id = current.comment.id;
          const noteBody = current.atom.querySelector('.report-comment-pop__field')?.value.trim() || '';
          const updated = await request(`/api/comments/${encodeURIComponent(id)}`, { method: 'PUT', body: JSON.stringify({ status: nextStatus, resolvedNote: nextStatus === 'resolved' ? noteBody : '' }) });
          close({ checkDirty: false });
          upsertComment(updated);
          setActiveComment(updated.id, { scroll: false });
        } catch (error) {
          message.textContent = error?.message || 'Could not update status.';
          setActionsBusy(false);
        }
      }

      async function save() {
        if (!current) return;
        const ta = current.atom.querySelector('.report-comment-pop__field');
        const message = current.atom.querySelector('.report-comment-pop__message');
        const body = (ta?.value || '').trim();
        if (!body) { message.textContent = 'Write a comment before saving.'; ta?.focus(); return; }
        setActionsBusy(true, 'Saving…');
        try {
          const result = current.mode === 'new'
            ? await request('/api/comments', { method: 'POST', body: JSON.stringify({ target: current.targetBlockId, body }) })
            : await request(`/api/comments/${encodeURIComponent(current.comment.id)}`, { method: 'PUT', body: JSON.stringify({ body }) });
          close({ checkDirty: false });
          upsertComment(result);
          setActiveComment(result.id, { scroll: false, updateHash: true });
        } catch (error) {
          message.textContent = error?.message || 'Could not save.';
          setActionsBusy(false);
        }
      }

      async function deleteCurrent() {
        if (!current || current.mode !== 'edit') return;
        if (!deleteConfirmed) return requestDeleteConfirm();
        deleteConfirmed = false;
        const message = current.atom.querySelector('.report-comment-pop__message');
        setActionsBusy(true, 'Deleting…');
        try {
          const id = current.comment.id;
          const deleted = await request(`/api/comments/${encodeURIComponent(id)}`, { method: 'DELETE' });
          close({ checkDirty: false });
          if (deleted?.id) { reviewComments = reviewComments.filter(c => c.id !== deleted.id); if (activeCommentId === deleted.id) activeCommentId = ''; renderInlineMarkers(); renderReviewTray(); }
        } catch (error) {
          message.textContent = error?.message || 'Could not delete.';
          setActionsBusy(false);
        }
      }

      function setActionsBusy(busy, label) {
        if (!current) return;
        const buttons = current.atom.querySelectorAll('.report-comment-pop__actions button');
        buttons.forEach(b => { b.disabled = !!busy; });
        const message = current.atom.querySelector('.report-comment-pop__message');
        if (busy && label) message.textContent = label;
        if (!busy && message.textContent && /…$/.test(message.textContent)) message.textContent = '';
      }

      // Close on rail/float breakpoint crossover. Dirty check still applies.
      if (typeof window !== 'undefined' && window.matchMedia) {
        window.matchMedia('(min-width: 1280px)').addEventListener('change', () => { if (current) close(); });
      }

      return { open, close, isDirty };
    })();

    function setReviewTrayOpen(open) {
      reviewTrayOpen = !!open;
      if (reviewTray) {
        reviewTray.classList.toggle('review-tray--open', reviewTrayOpen);
        reviewTray.setAttribute('aria-hidden', String(!reviewTrayOpen));
      }
      if (reviewRail) {
        reviewRail.setAttribute('aria-expanded', String(reviewTrayOpen));
        reviewRail.setAttribute('aria-label', reviewTrayOpen ? 'Hide review comments' : 'Open review comments');
      }
    }

    function toggleReviewTray() {
      setReviewTrayOpen(!reviewTrayOpen);
    }

    function closeReviewTray() {
      setReviewTrayOpen(false);
    }

    function updateReviewRailCounts(open, total) {
      document.querySelectorAll('[data-review-rail-open-count]').forEach(node => {
        node.textContent = String(open);
      });
      document.querySelectorAll('[data-review-rail-total-count]').forEach(node => {
        node.textContent = String(total);
      });
    }

    function setCommentMode(on) {
      document.body.dataset.commentMode = String(!!on);
      if (!on) popup.close({ checkDirty: false });
      if (!commentToggle) return;
      commentToggle.classList.toggle('is-active', !!on);
      commentToggle.setAttribute('aria-pressed', String(!!on));
      commentToggle.setAttribute('aria-label', on ? 'Stop comment mode' : 'Start comment mode');
      commentToggle.setAttribute('title', on ? 'Stop comment mode' : 'Start comment mode');
    }

    function toggleCommentMode() {
      setCommentMode(document.body.dataset.commentMode !== 'true');
    }

    setCommentMode(false);
    if (commentToggle) {
      commentToggle.addEventListener('click', toggleCommentMode);
    }
    if (bannerClose) bannerClose.addEventListener('click', () => setCommentMode(false));
    if (reviewRail) reviewRail.addEventListener('click', toggleReviewTray);

    async function request(path, options = {}) {
      const res = await fetch(path, {
        headers: { 'content-type': 'application/json', ...(options.headers || {}) },
        ...options,
      });
      if (!res.ok) {
        const text = await res.text();
        throw new Error(text || `Request failed: ${res.status}`);
      }
      return res.headers.get('content-type')?.includes('application/json') ? res.json() : res.text();
    }

    function escapeText(value) {
      return String(value || '').replace(/[&<>"']/g, c => ({ '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;' }[c]));
    }
    function openCommentEditor(comment) {
      const block = comment.target ? findBlock(comment.target) : null;
      popup.open({ mode: 'edit', comment, originEl: block });
    }

    function openResolveDialog(comment) {
      // Resolve / reopen happen inside the popup via the status pill.
      const block = comment.target ? findBlock(comment.target) : null;
      popup.open({ mode: 'edit', comment, originEl: block });
    }

    async function reopenComment(comment) {
      const updated = await request(`/api/comments/${encodeURIComponent(comment.id)}`, {
        method: 'PUT',
        body: JSON.stringify({ status: 'open', resolvedNote: '' }),
      });
      upsertComment(updated);
      setActiveComment(updated.id, { scroll: false });
    }

    function upsertComment(comment) {
      const index = reviewComments.findIndex(item => item.id === comment.id);
      if (index >= 0) reviewComments[index] = comment;
      else reviewComments.push(comment);
      renderInlineMarkers();
      renderReviewTray();
    }

    function filteredComments() {
      if (activeFilter === 'open' || activeFilter === 'resolved') {
        return reviewComments.filter(comment => (comment.status || 'open') === activeFilter);
      }
      return reviewComments;
    }

    function blockIdFromTarget(target) {
      return String(target || '').replace(/^block:/u, 'block-');
    }
