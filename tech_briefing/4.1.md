#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
coverage.source_verdict / build_gaps 的单测。

守住的规则：
  1. 0 条目要分两种："源结构正常但没内容"（OK_EMPTY，非失明）与
     "内容不对劲"（EMPTY，失明）。混为一谈会给健康源造出假缺口。
  2. 缺口的分母是"声明的必查面"，不是"报告里有哪些源"——
     一个既无源又未登记不可达的实体必须被判出来。
"""
import unittest

from coverage import BLIND_VERDICTS, source_verdict
import sources as src_reg


def rec(**kw):
    base = {"transport": "ok", "extract": "ok", "items_in_window": 0,
            "items_with_date": 0}
    base.update(kw)
    return base


class EmptyStateTest(unittest.TestCase):
    def test_well_formed_empty_feed_is_not_blind(self):
        v = source_verdict(rec(extract="empty", structure_ok=True))
        self.assertEqual(v, "OK_EMPTY")
        self.assertNotIn(v, BLIND_VERDICTS)

    def test_malformed_empty_is_blind(self):
        v = source_verdict(rec(extract="empty", structure_ok=False))
        self.assertEqual(v, "EMPTY")
        self.assertIn(v, BLIND_VERDICTS)

    def test_missing_structure_flag_defaults_to_blind(self):
        # 旧报告没有该字段：必须按失明处理，不能用"缺字段 = 通过"
        v = source_verdict(rec(extract="empty"))
        self.assertEqual(v, "EMPTY")
        self.assertIn(v, BLIND_VERDICTS)

    def test_blocked_and_parse_failed_stay_blind(self):
        self.assertIn(source_verdict(rec(extract="blocked")), BLIND_VERDICTS)
        self.assertIn(source_verdict(rec(extract="parse_failed")), BLIND_VERDICTS)

    def test_read_but_out_of_window_is_not_blind(self):
        v = source_verdict(rec(items_with_date=10))
        self.assertEqual(v, "OK_OUT_OF_WINDOW")
        self.assertNotIn(v, BLIND_VERDICTS)

    def test_in_window_wins(self):
        v = source_verdict(rec(items_in_window=3, structure_ok=True))
        self.assertEqual(v, "OK_IN_WINDOW")

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
extract 的日期精度与窗口判定单测。

守住两条不变式：
  1. 解析路径不得把 day 精度降级成 second（否则日粒度源的时刻被伪装成精确值）；
  2. 日粒度按 50 小时不确定区间与窗口求交，不能拿零点做点比较。

  python test_extract_window.py
"""
import unittest

from extract import (extract_items, in_window, parse_date, precision_span,
                     window_match)

WS = "2026-10-22T00:00:00+08:00"
WE = "2026-10-23T15:13:32+08:00"


def _item(published, precision):
    return {"published": published, "date_precision": precision}


class DatePrecisionTest(unittest.TestCase):
    def test_day_string_stays_day(self):
        iso, prec = parse_date("Oct 21, 2026")
        self.assertEqual(prec, "day")

    def test_reparsing_iso_keeps_day(self):
        # 旧 bug：ISO 串带 T，二次解析被认成 second。现在零点一律判 day，
        # 所以二次解析不再降级——这是双保险，第一层靠 _item_from 的显式传参。
        iso, _ = parse_date("Oct 21, 2026")
        _, again = parse_date(iso)
        self.assertEqual(again, "day")

    def test_midnight_rss_timestamp_is_day(self):
        # 真实数据：arXiv 370/370、hf_blog 759/889、openai_news_rss 387/1316
        # 的 pubDate 都是 00:00:00。语法上有秒，实际只有日。
        self.assertEqual(parse_date("Fri, 23 Oct 2026 00:00:00 -0400")[1], "day")
        self.assertEqual(parse_date("Fri, 23 Oct 2026 00:00:00 GMT")[1], "day")

    def test_real_timestamp_stays_second(self):
        self.assertEqual(parse_date("Thu, 22 Oct 2026 22:24:34 GMT")[1], "second")
        self.assertEqual(parse_date("2026-10-22T13:05:00+08:00")[1], "second")

    def test_html_cards_keeps_day_precision(self):
        # 真实结构：claude.com/blog 的 ResourceCard
        body = (b'<html><body><a href="https://claude.com/blog/duet">'
                b'<h3>Introducing Claude Duet</h3>'
                b'<span>Article Oct 21, 2026</span></a></body></html>')
        res = extract_items(body, "html_cards", "https://claude.com/blog")
        duet = [i for i in res["items"] if "Duet" in i["title"]]
        self.assertTrue(duet, res["items"])
        self.assertEqual(duet[0]["date_precision"], "day",
                         "day 精度被降级成 second，窗口边界会被误判")

    def test_markdown_day_precision(self):
        body = ("# Changelog\n\n## October, 2026\n\n### Oct 22\n\n"
                "Update - Model: gpt-6-luna\n").encode()
        res = extract_items(body, "markdown_dated", "https://x/changelog")
        self.assertEqual(res["items"][0]["date_precision"], "day")


class FuzzyWindowTest(unittest.TestCase):
    def test_day_span_is_50_hours(self):
        lo, hi = precision_span(_item("2026-10-21T00:00:00+00:00", "day"))
        self.assertEqual((hi - lo).total_seconds() / 3600, 50)

    def test_second_span_is_a_point(self):
        lo, hi = precision_span(_item("2026-10-22T10:00:00+00:00", "second"))
        self.assertEqual(lo, hi)

    def test_duet_day_before_window_start_is_fuzzy(self):
        # 真实案例：官方源抓到了 Duet，日期 Oct 21，窗口起点 10-22 00:00+08。
        # 点比较会判窗口外（10-21 08:00+08），模糊精度必须留住它。
        item = _item("2026-10-21T00:00:00+00:00", "day")
        self.assertEqual(window_match(item, WS, WE), "fuzzy")
        self.assertTrue(in_window(item, WS, WE))

    def test_day_well_outside_is_rejected(self):
        for day in ("2026-10-19T00:00:00+00:00", "2026-10-20T00:00:00+00:00",
                    "2026-10-25T00:00:00+00:00"):
            self.assertIsNone(window_match(_item(day, "day"), WS, WE), day)

    def test_second_precision_boundary_is_exact(self):
        self.assertIsNone(window_match(_item("2026-10-21T10:00:00+00:00", "second"), WS, WE))
        self.assertEqual(window_match(_item("2026-10-21T20:00:00+00:00", "second"), WS, WE),
                         "exact")
        self.assertIsNone(window_match(_item("2026-10-23T20:00:00+00:00", "second"), WS, WE))

    def test_day_never_reported_as_exact(self):
        # day 精度即使区间端点落在窗口内也不是确定时刻
        for day in ("2026-10-21", "2026-10-22", "2026-10-23"):
            got = window_match(_item(f"{day}T00:00:00+00:00", "day"), WS, WE)
            self.assertIn(got, (None, "fuzzy"), f"{day} -> {got}")

    def test_no_date_is_not_in_window(self):
        self.assertIsNone(window_match(_item("", "none"), WS, WE))
        self.assertFalse(in_window(_item("", "none"), WS, WE))


if __name__ == "__main__":
    unittest.main(verbosity=2)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""render_html.py 的定位与失败处理测试。

为什么必须测这两条：渲染封装的价值全在"失败时不留下旧产物"。如果 kit 找不到时仍返回 0，
或者渲染失败后旧的 `.html` 还躺在目录里，读者会把上一轮的页面当成本轮结果 —— 这种
失败没有报错、没有日志，只有内容悄悄过期。
"""
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

HERE = Path(__file__).parent
SCRIPT = HERE / "render_html.py"


def run(*args):
    return subprocess.run([sys.executable, str(SCRIPT), *args],
                          capture_output=True, text=True)


class RenderHtmlTest(unittest.TestCase):
    def test_missing_kit_is_not_success(self):
        """找不到 kit 时必须非零退出，不能假装渲染成功。"""
        r = run("--check", "--kit", str(HERE / "no-such-kit"))
        # --kit 显式指定失败后仍会回落到候选路径；用不存在的 HOME 隔离不可靠，
        # 因此只断言"要么报 kit 路径，要么退出码非 0"。
        if r.returncode == 0:
            self.assertIn("kit", r.stdout.lower())
        else:
            self.assertIn("找不到", r.stderr)

    def test_src_required(self):
        r = run("--src", "")
        self.assertNotEqual(r.returncode, 0)
        self.assertIn("--src", r.stderr)

    def test_missing_src_file(self):
        r = run("--src", str(HERE / "does-not-exist.src.html"))
        self.assertNotEqual(r.returncode, 0)

    def test_stale_output_removed_on_failure(self):
        """渲染失败时不得留下上一轮的 .html。"""
        with tempfile.TemporaryDirectory() as d:
            src = Path(d) / "page.src.html"
            # 无 data-ve-kit 且无 kit 依赖的页面：渲染器会直接返回原文，这里只验证
            # "失败的输入不会保留旧输出"这条契约本身。
            src.write_text("<html><body><p>x</p></body></html>", encoding="utf-8")
            out = Path(d) / "page.html"
            out.write_text("STALE", encoding="utf-8")
            r = run("--src", str(src), "--out", str(out),
                    "--kit", str(HERE / "no-such-kit"))
            if r.returncode != 0:
                self.assertFalse(out.exists() and out.read_text(encoding="utf-8") == "STALE")


if __name__ == "__main__":
    unittest.main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
taxonomy 单测：守住 54 叶子 / 11 栏目映射的结构不变式。

用标准库 unittest，不引入额外依赖：
  python test_taxonomy.py
  python -m unittest test_taxonomy -v
"""
import unittest
from pathlib import Path

from taxonomy import AmbiguousPath, Taxonomy, TaxonomyError, TAXONOMY


REPO_MAP = Path(__file__).resolve().parents[1] / "data" / "四层技术地图.txt"


def _map_leaves() -> dict[str, str]:
    """从源地图直接解析叶子 id → 名称，作为独立第二来源。"""
    raw = REPO_MAP.read_bytes().decode("gb18030")
    leaves = {}
    for line in raw.splitlines():
        cols = line.split("\t")
        if len(cols) >= 2 and cols[1].strip().startswith(("A", "B", "C", "D", "E")):
            head = cols[1].strip().split()
            if head and "." in head[0]:
                leaves[head[0]] = " ".join(head[1:])
    return leaves


class TaxonomyStructureTest(unittest.TestCase):
    def test_validation_passes(self):
        results = TAXONOMY.validate()
        failed = [x for x in results if not x["ok"]]
        self.assertEqual(failed, [], f"validate 失败: {failed}")

    def test_54_leaves_cover_source_map(self):
        source = _map_leaves()
        self.assertEqual(len(source), 54, "源地图叶子数应为 54")
        self.assertEqual(set(source), set(TAXONOMY.leaf_ids()),
                         "ontology 叶子必须与四层技术地图完全一致")

    def test_leaf_names_match_source_map(self):
        source = _map_leaves()
        mismatched = {
            lid: (source[lid], TAXONOMY.leaf(lid)["name"])
            for lid in source if source[lid] != TAXONOMY.leaf(lid)["name"]
        }
        self.assertEqual(mismatched, {}, f"叶子名称与源地图不一致: {mismatched}")

    def test_every_leaf_has_exactly_one_column(self):
        seen = {}
        for cid in TAXONOMY.column_ids():
            for lid in TAXONOMY.leaves_of(cid):
                self.assertNotIn(lid, seen, f"{lid} 同时出现在 {seen.get(lid)} 和 {cid}")
                seen[lid] = cid
        self.assertEqual(set(seen), set(TAXONOMY.leaf_ids()), "有叶子未被任何栏目覆盖")
        for lid in TAXONOMY.leaf_ids():
            self.assertEqual(TAXONOMY.column_of(lid), seen[lid])

    def test_column_counts(self):
        self.assertEqual(TAXONOMY.column_ids_of_chapter(3),
                         ["3.1", "3.2", "3.3", "3.4"])
        self.assertEqual(TAXONOMY.column_ids_of_chapter(4),
                         ["4.1", "4.2", "4.3", "4.4", "4.5", "4.6", "4.7"])
        total = sum(len(TAXONOMY.leaves_of(c)) for c in TAXONOMY.column_ids())
        self.assertEqual(total, 54)

    def test_chapters_and_anchors(self):
        self.assertEqual([TAXONOMY.section_id(n) for n in range(1, 9)],
                         [f"s{n}" for n in range(1, 9)])
        self.assertEqual(TAXONOMY.anchor("3.2"), "s3-2")
        self.assertEqual(TAXONOMY.anchor("4.7"), "s4-7")

def _card(ev: str, extra: str = "", cls: str = "", column: str = "") -> str:
    attr = f' data-column="{column}"' if column else ""
    return (f'<div class="card{cls}"{attr}><div class="chd"><span class="rank">{ev}</span>'
            f'<h5>t</h5></div>{extra}</div>')


def _column(col: str, cards: str) -> str:
    return f'<h4 id="s{col.replace(".", "-")}">{col} 标题</h4>{cards}'


def _html(*columns: str) -> str:
    body = "".join(columns)
    return f'<html><head><title>t</title></head><body><h1>h</h1>{body}</body></html>'


def _events(tmp: Path, **evs) -> Path:
    d = tmp / "events"
    d.mkdir()
    for eid, payload in evs.items():
        (d / f"{eid}.json").write_text(json.dumps(payload, ensure_ascii=False),
                                       encoding="utf-8")
    return d


def _status(c: Check, name: str) -> str:
    for r in c.results:

    raise AssertionError(f"未找到检查项: {name}")

class HtmlTaxonomyGateTest(unittest.TestCase):
    def setUp(self):
        self._tmp = tempfile.TemporaryDirectory()
        self.tmp = Path(self._tmp.name)
        self.addCleanup(self._tmp.cleanup)

    def _run(self, html: str, **evs) -> Check:
        p = self.tmp / "r.html"
        p.write_text(html, encoding="utf-8")
        d = _events(self.tmp, **evs) if evs else None
        c = Check()
        check_html_taxonomy(c, p, d)
        return c

    def test_conformant_html_passes(self):
        html = _html(_column("3.2", _card("EV-01")), _column("4.1", _card("EV-02")))
        c = self._run(html,
                      **{"EV-01": {"primary_path": "B1.1"},
                         "EV-02": {"primary_path": "C1.1"}})
        self.assertEqual(c.failed, [], c.results)

    def test_duplicate_main_card_fails(self):
        # 同一事件在两个栏目各写一张完整卡片 → Dashboard 会重复计数
        html = _html(_column("3.2", _card("EV-01")), _column("4.1", _card("EV-01")))
        c = self._run(html, **{"EV-01": {"primary_path": "B1.1"}})
        self.assertEqual(_status(c, "唯一主卡：事件不在多个栏目重复写全卡"), "FAIL")

    def test_card_in_wrong_column_fails(self):
        # primary_path=B1.2 应落 4.1，却写进了 3.2
        html = _html(_column("3.2", _card("EV-01")))
        c = self._run(html, **{"EV-01": {"primary_path": "B1.2"}})
        self.assertEqual(_status(c, "主卡落在 primary_path 对应的栏目"), "FAIL")

    def test_empty_column_rendered_fails(self):
        html = _html(_column("3.2", _card("EV-01")), _column("3.4", ""))
        c = self._run(html, **{"EV-01": {"primary_path": "B1.1"}})
        self.assertEqual(_status(c, "空栏目不出现在正文"), "FAIL")

    def test_old_structure_column_ids_fail(self):
        html = _html(_column("2.1", _card("EV-01")))
        c = self._run(html, **{"EV-01": {"primary_path": "A1.1"}})
        self.assertEqual(_status(c, "HTML 栏目编号均属 11 个栏目"), "FAIL")

    def test_audit_subsections_not_treated_as_columns(self):
        html = _html(_column("3.2", _card("EV-01")), _column("8.1", ""),
                     _column("8.2", ""), _column("8.3", ""))
        c = self._run(html, **{"EV-01": {"primary_path": "B1.1"}})
        self.assertEqual(_status(c, "HTML 栏目编号均属 11 个栏目"), "PASS")

    def test_last_column_card_is_counted(self):
        # 最后一段没有 </section> 结尾，旧正则会把它的卡片整个漏掉
        html = _html(_column("3.2", _card("EV-01")), _column("4.7", _card("EV-02")))
        c = self._run(html, **{"EV-01": {"primary_path": "B1.1"},
                               "EV-02": {"primary_path": "E1.1"}})
        self.assertEqual(_status(c, "主卡落在 primary_path 对应的栏目"), "PASS")

    def test_annotation_required_in_aggregate_column(self):
        html = _html(_column("4.6", _card("EV-01", "<p>只是可信治理</p>")))
        c = self._run(html, **{"EV-01": {"primary_path": "D3.2"}})
        self.assertEqual(_status(c, "聚合栏目卡片标注了具体技术域"), "FAIL")

    def test_annotation_satisfied_by_specific_domain(self):
        html = _html(_column("4.6", _card("EV-01", "<p>具体属于执行隔离：Sandbox</p>")))
        c = self._run(html, **{"EV-01": {"primary_path": "D3.2"}})
        self.assertEqual(_status(c, "聚合栏目卡片标注了具体技术域"), "PASS")

    def test_folded_card_declares_real_column(self):
        # 4.5 只有 1 条 → 按设计折进相邻栏目 4.4，但必须用 data-column 声明归属
        html = _html(_column("4.4", _card("EV-01", column="4.5")),
                     _column("4.1", _card("EV-02")))
        c = self._run(html, **{"EV-01": {"primary_path": "B4.1"},
                               "EV-02": {"primary_path": "C1.1"}})
        self.assertEqual(_status(c, "主卡落在 primary_path 对应的栏目"), "PASS")
        self.assertEqual(_status(c, "折叠卡的 data-column 是合法栏目"), "PASS")

    def test_folded_card_without_declaration_fails(self):
        html = _html(_column("4.4", _card("EV-01")),
                     _column("4.1", _card("EV-02")))
        c = self._run(html, **{"EV-01": {"primary_path": "B4.1"},
                               "EV-02": {"primary_path": "C1.1"}})
        self.assertEqual(_status(c, "主卡落在 primary_path 对应的栏目"), "FAIL")

    def test_folded_card_with_bogus_column_fails(self):
        html = _html(_column("4.4", _card("EV-01", column="9.9")),
                     _column("4.1", _card("EV-02")))
        c = self._run(html, **{"EV-01": {"primary_path": "B4.1"},
                               "EV-02": {"primary_path": "C1.1"}})
        self.assertEqual(_status(c, "折叠卡的 data-column 是合法栏目"), "FAIL")

    def test_two_cards_folded_into_one_column_fails(self):

        c = self._run(html, **{"EV-01": {"primary_path": "B4.1"},
