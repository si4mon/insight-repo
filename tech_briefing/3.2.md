
    function findBlock(target) {
      return document.querySelector(`[data-report-block-id="${CSS.escape(target)}"]`);
    }

    let reportCommentLayoutFrame = 0;
    let reportCommentResizeObserver = null;
    function getReportCommentResizeObserver() {
      if (reportCommentResizeObserver || typeof ResizeObserver === 'undefined') return reportCommentResizeObserver;
      reportCommentResizeObserver = new ResizeObserver(scheduleReportCommentLayout);
      return reportCommentResizeObserver;
    }
    function resetReportCommentMarginStyles(note) {
      note.style.position = '';
      note.style.float = '';
      note.style.clear = '';
      note.style.left = '';
      note.style.top = '';
      note.style.width = '';
      note.style.margin = '';
      note.style.transform = '';
      note.style.visibility = '';
    }

    function layoutReportCommentMargins() {
      reportCommentLayoutFrame = 0;
      const notes = Array.from(document.querySelectorAll('.report-comment-margin'));
      if (!notes.length) return;
      if (!article || !window.matchMedia('(min-width: 1180px)').matches) {
        notes.forEach(resetReportCommentMarginStyles);
        return;
      }

      notes.forEach(note => {
        note.style.position = 'absolute';
        note.style.float = 'none';
        note.style.clear = 'none';
        note.style.left = 'calc(100% + var(--s-8))';
        note.style.top = '0px';
        note.style.width = '15rem';
        note.style.margin = '0';
        note.style.transform = 'none';
        note.style.visibility = 'hidden';
      });

      const articleRect = article.getBoundingClientRect();
      const placements = notes.map(note => {
        const block = findBlock(note.dataset.reportCommentTarget);
        const blockRect = block ? block.getBoundingClientRect() : articleRect;
        return {
          note,
          top: Math.max(0, Math.round(blockRect.top - articleRect.top)),
        };
      }).sort((a, b) => a.top - b.top);

      let nextTop = 0;
      placements.forEach(({ note, top }) => {
        const placedTop = Math.max(top, nextTop);
        note.style.top = `${placedTop}px`;
        note.style.visibility = '';
        nextTop = placedTop + note.offsetHeight + 16;
      });
    }

    function scheduleReportCommentLayout() {
      if (reportCommentLayoutFrame) return;
      reportCommentLayoutFrame = window.requestAnimationFrame(layoutReportCommentMargins);
    }

    function renderInlineMarkers() {
      document.querySelectorAll('.report-comment-margin').forEach(node => node.remove());
      if (article) {
        article.querySelectorAll('.table-wrap[data-report-has-comment="true"]').forEach(tableWrap => {
          delete tableWrap.dataset.reportHasComment;
        });
      }
      const useDesktopMargin = article && window.matchMedia('(min-width: 1180px)').matches;
      const groups = new Map();
      reviewComments.forEach(comment => {
        if (!comment.target) return;
        if (!groups.has(comment.target)) groups.set(comment.target, []);
        groups.get(comment.target).push(comment);
      });
      groups.forEach((comments, target) => {
        const block = findBlock(target);
        if (!block) return;
        const enclosingTable = block.closest('.table-wrap');
        if (enclosingTable) enclosingTable.dataset.reportHasComment = 'true';
        const wrap = document.createElement('aside');
        wrap.className = 'report-comment-margin';
        wrap.setAttribute('aria-label', `Review comments for ${target}`);
        wrap.dataset.reportCommentTarget = target;
        comments.forEach(comment => {
          const status = comment.status || 'open';
          const card = document.createElement('article');
          card.className = 'report-comment-note';
          card.setAttribute('data-report-comment-note', comment.id);
          card.dataset.reportCommentStatus = status;
          const marker = document.createElement('a');
          marker.className = 'report-comment-note__meta';
          marker.id = `comment-${comment.id}`;
          marker.href = `#comment-${comment.id}`;
          marker.setAttribute('data-report-comment-marker', comment.id);
          marker.dataset.reportCommentStatus = status;
          marker.innerHTML = `<span class="report-comment-note__status">${escapeText(status)}</span><span class="report-comment-note__id">${escapeText(comment.id)}</span>`;
          marker.setAttribute('aria-label', `${status} comment ${comment.id}`);
          marker.addEventListener('click', event => {
            event.preventDefault();
            setActiveComment(comment.id, { updateHash: true });
          });
          const body = document.createElement('p');
          body.className = 'report-comment-note__body';
          body.textContent = comment.body || 'No comment text.';
          card.appendChild(marker);
          card.appendChild(body);
          if (comment.resolvedNote) {
            const resolved = document.createElement('p');
            resolved.className = 'report-comment-note__resolved';
            resolved.textContent = `Resolved: ${comment.resolvedNote}`;
            card.appendChild(resolved);
          }
          wrap.appendChild(card);
        });
        if (useDesktopMargin) article.appendChild(wrap);
        else (block.closest('.table-wrap, ul, ol, dl') || block).insertAdjacentElement('afterend', wrap);
        const observer = getReportCommentResizeObserver();
        if (observer) observer.observe(wrap);
      });
      syncActiveClasses();
      scheduleReportCommentLayout();
    }

    let reportCommentRenderFrame = 0;
    function scheduleRenderInlineMarkers() {
      if (reportCommentRenderFrame) return;
      reportCommentRenderFrame = window.requestAnimationFrame(() => {
        reportCommentRenderFrame = 0;
        renderInlineMarkers();
      });
    }

    function renderReviewTray() {
      if (!reviewTray) return;
      const open = reviewComments.filter(comment => (comment.status || 'open') !== 'resolved').length;
      const resolved = reviewComments.filter(comment => (comment.status || 'open') === 'resolved').length;
      const comments = filteredComments();
      const active = reviewComments.find(comment => comment.id === activeCommentId) || comments[0] || null;
      if (!activeCommentId && active) activeCommentId = active.id;
      updateReviewRailCounts(open, reviewComments.length);
      reviewTray.innerHTML = `<div class="review-tray__header">
          <div>
            <h2 class="review-tray__title">Review</h2>
            <p class="review-tray__counts">Open ${open} · Resolved ${resolved} · Total ${reviewComments.length}</p>
          </div>
          <button class="review-tray__close" type="button" data-action="close-tray" aria-label="Hide review comments">×</button>
        </div>
        <div class="review-tray__filter" role="group" aria-label="Filter review comments">
          ${['open', 'resolved', 'all'].map(filter => `<button type="button" data-filter="${filter}" aria-pressed="${String(activeFilter === filter)}">${filter[0].toUpperCase() + filter.slice(1)}</button>`).join('')}
        </div>
        <div class="review-tray__list">
          ${comments.length ? comments.map(comment => renderReviewTrayItem(comment)).join('') : '<p class="review-tray__empty">No comments in this view.</p>'}
        </div>`;

      reviewTray.querySelectorAll('[data-action="close-tray"]').forEach(button => {
        button.addEventListener('click', closeReviewTray);
      });
      reviewTray.querySelectorAll('[data-filter]').forEach(button => {
        button.addEventListener('click', () => {
          activeFilter = button.dataset.filter || 'open';
          activeCommentId = '';
          renderReviewTray();
          syncActiveClasses();
        });
      });
      reviewTray.querySelectorAll('[data-comment-id]').forEach(row => {
        row.addEventListener('click', event => {
          if (event.target.closest('a, button')) return;
          setActiveComment(row.dataset.commentId, { updateHash: true });
        });
      });
      reviewTray.querySelectorAll('[data-action="edit"]').forEach(button => {
        button.addEventListener('click', () => {
          const comment = reviewComments.find(item => item.id === button.dataset.commentId);
          if (comment) openCommentEditor(comment);
        });
      });
      reviewTray.querySelectorAll('[data-action="resolve"]').forEach(button => {
        button.addEventListener('click', () => {
          const comment = reviewComments.find(item => item.id === button.dataset.commentId);
          if (comment) openResolveDialog(comment);
        });
      });
      reviewTray.querySelectorAll('[data-action="reopen"]').forEach(button => {
        button.addEventListener('click', () => {
          const comment = reviewComments.find(item => item.id === button.dataset.commentId);
          if (comment) reopenComment(comment).catch(error => showToast(escapeText(error.message || 'Could not reopen comment.')));
        });
      });
      reviewTray.querySelectorAll('[data-action="delete"]').forEach(button => {
        button.addEventListener('click', () => {
          const comment = reviewComments.find(item => item.id === button.dataset.commentId);
          if (comment) openCommentEditor(comment);
        });
      });
      syncActiveClasses();
    }

    function renderReviewTrayItem(comment) {
      const status = comment.status || 'open';
      const blockHash = `#${blockIdFromTarget(comment.target)}`;
      const commentHash = `#comment-${comment.id}`;
      const active = comment.id === activeCommentId ? ' is-active' : '';
      const body = escapeText(comment.body || '').slice(0, 220);
      const resolveAction = status === 'resolved'
        ? `<button class="review-tray__action" type="button" data-action="reopen" data-comment-id="${escapeText(comment.id)}">Reopen</button>`
        : `<button class="review-tray__action" type="button" data-action="resolve" data-comment-id="${escapeText(comment.id)}">Resolve</button>`;
      return `<div class="review-tray__item${active}" data-comment-id="${escapeText(comment.id)}" tabindex="0">
        <div class="review-tray__item-meta">
          <span>${escapeText(comment.id)} · ${escapeText(status)}</span>
          <span>${escapeText(comment.target || '')}</span>
        </div>
        <p class="review-tray__item-body">${body || 'No comment text.'}</p>
        ${comment.resolvedNote ? `<p class="review-tray__item-body"><strong>Resolved:</strong> ${escapeText(comment.resolvedNote)}</p>` : ''}
        <div class="review-tray__links">
          <a href="${commentHash}">Comment link</a>
          <a href="${blockHash}">Block link</a>
        </div>
        <div class="review-tray__actions">
          <button class="review-tray__action" type="button" data-action="edit" data-comment-id="${escapeText(comment.id)}">Edit</button>
          ${resolveAction}
          <button class="review-tray__action" type="button" data-action="delete" data-comment-id="${escapeText(comment.id)}">Delete</button>
        </div>
      </div>`;
    }

    function syncActiveClasses() {
      document.querySelectorAll('.report-comment-note__meta').forEach(marker => {
        marker.classList.toggle('is-active', marker.dataset.reportCommentMarker === activeCommentId);
      });
      document.querySelectorAll('[data-report-comment-note]').forEach(card => {
        card.classList.toggle('is-active', card.getAttribute('data-report-comment-note') === activeCommentId);
      });
      document.querySelectorAll('[data-report-block-id]').forEach(block => {
        const active = reviewComments.find(comment => comment.id === activeCommentId);
        block.classList.toggle('is-review-active', !!active && block.dataset.reportBlockId === active.target);
      });
    }

    function setActiveComment(id, options = {}) {
      const { scroll = true, updateHash = false } = options;
      const comment = reviewComments.find(item => item.id === id);
      if (!comment) return;
      activeCommentId = comment.id;
      renderReviewTray();
      syncActiveClasses();
      if (scroll) {
        const block = findBlock(comment.target);
        if (block) block.scrollIntoView({ behavior: 'smooth', block: 'center' });
      }
      if (updateHash) history.replaceState(null, '', `#comment-${comment.id}`);
    }

    function jumpToNextOpenComment() {
      const open = reviewComments.filter(comment => (comment.status || 'open') !== 'resolved');
      if (!open.length) return;
      const index = open.findIndex(comment => comment.id === activeCommentId);
      const next = open[(index + 1) % open.length];
      setActiveComment(next.id, { updateHash: true });
    }

    function flashReviewTarget(block) {
      if (!block) return;
      block.classList.remove('is-review-flash');
      void block.offsetWidth;
      block.classList.add('is-review-flash');
      setTimeout(() => block.classList.remove('is-review-flash'), 1400);
    }

    function handleReviewHash() {
      const hash = decodeURIComponent(window.location.hash || '');
      if (hash.startsWith('#comment-')) {
        const id = hash.replace('#comment-', '');
        const comment = reviewComments.find(item => item.id === id);
        if (comment) {
          setActiveComment(id);
          flashReviewTarget(findBlock(comment.target));
        }
        return;
      }
      if (hash.startsWith('#block-')) {
        const target = hash.replace('#block-', 'block:');
        const block = findBlock(target);
        if (block) {
          block.scrollIntoView({ behavior: 'smooth', block: 'center' });
          flashReviewTarget(block);
        }
      }
    }

    function isTypingTarget(target) {
      return !!target.closest('input, textarea, select, [contenteditable="true"]');
    }

    document.addEventListener('keydown', event => {
      if (event.key === 'Escape') {
        // The popup owns its own Esc inside bindKeys; if it is open we let it
        // handle the event. Otherwise we exit comment mode.
        if (document.querySelector('.report-comment-pop')) return;
        setCommentMode(false);
        return;
      }
      if (isTypingTarget(event.target)) return;
      if (event.altKey && event.code === 'KeyC') {
        event.preventDefault();
        toggleCommentMode();
      }
      if (event.key === 'n') {
        event.preventDefault();
        jumpToNextOpenComment();
      }
    });

    document.querySelectorAll('[data-report-block-id]').forEach(block => {
      block.setAttribute('title', `Turn on comment mode to comment on ${block.dataset.reportBlockId}`);
      block.addEventListener('click', async (event) => {
        if (event.target.closest('a, button, input, textarea, select, .report-comment-margin')) return;
        if (document.body.dataset.commentMode !== 'true') return;
        const target = block.dataset.reportBlockId;
        const firstUnresolved = reviewComments.find(c => c.target === target && c.status !== 'resolved');
        if (firstUnresolved) popup.open({ mode: 'edit', comment: firstUnresolved, originEl: block });
        else popup.open({ mode: 'new', target, originEl: block });
      });
    });

    window.addEventListener('resize', scheduleRenderInlineMarkers, { passive: true });
    window.addEventListener('load', scheduleReportCommentLayout, { once: true });
    if (document.fonts?.ready) document.fonts.ready.then(scheduleReportCommentLayout).catch(() => {});

    request('/api/comments').then(comments => {
      reviewComments = comments;
      renderInlineMarkers();
      renderReviewTray();
      handleReviewHash();
    }).catch(() => {});
    window.addEventListener('hashchange', handleReviewHash);
  }

  initReviewMode();

  const toast = document.getElementById('toast');
  let toastTimer = null;
  function showToast(html) {
    toast.innerHTML = html;
    toast.setAttribute('data-visible', 'true');
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => toast.setAttribute('data-visible', 'false'), 2500);
  }

  // / = focus first ToC link
  // t = toggle theme
  // s = save as MD
  // g 1..9 = jump to nth top-level section
  /* ---------- Lightbox ---------- */
  const lightbox = document.getElementById('lightbox');
  const lightboxStage = document.getElementById('lightbox-stage');
  const lightboxCaption = document.getElementById('lightbox-caption');
  const lightboxClose = document.getElementById('lightbox-close');
  let lightboxLastFocus = null;

  function openLightboxFor(target) {
    const fig = target.closest('figure');
    const cap = fig ? fig.querySelector('figcaption') : null;
    // Clone the visible content (img or svg) into the stage.
    const content = target.querySelector('img, svg') || (target.matches('img, svg') ? target : null);
    if (!content) return;
    lightboxLastFocus = document.activeElement;
    lightboxStage.innerHTML = '';
    const clone = content.cloneNode(true);
    if (clone.tagName === 'IMG') {
      clone.removeAttribute('width');
      clone.removeAttribute('height');
    }
    lightboxStage.appendChild(clone);
    lightboxCaption.textContent = cap ? cap.textContent.trim() : '';
    lightbox.setAttribute('data-open', 'true');
    document.documentElement.style.overflow = 'hidden';
    requestAnimationFrame(() => lightboxClose.focus());
  }

  function closeLightbox() {
    lightbox.setAttribute('data-open', 'false');
    document.documentElement.style.overflow = '';
    setTimeout(() => { lightboxStage.innerHTML = ''; }, 240);
    if (lightboxLastFocus && typeof lightboxLastFocus.focus === 'function') {
      lightboxLastFocus.focus();
    }
  }

  document.querySelectorAll('figure.framed .zoomable').forEach(target => {
    if (!target.hasAttribute('tabindex')) target.setAttribute('tabindex', '0');
    if (!target.hasAttribute('role')) target.setAttribute('role', 'button');
    target.addEventListener('click', () => openLightboxFor(target));
    target.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); openLightboxFor(target); }
    });
  });

  lightbox.addEventListener('click', (e) => {
    if (e.target === lightbox
        || e.target.classList.contains('lightbox__inner')
        || e.target.classList.contains('lightbox__stage')) closeLightbox();
  });
  lightboxClose.addEventListener('click', closeLightbox);

  let gPrefix = false;
  let gTimer = null;

  document.addEventListener('keydown', (e) => {
    if (lightbox.getAttribute('data-open') === 'true' && e.key === 'Escape') {
      e.preventDefault();
      closeLightbox();
      return;
    }

    // Alt/Option-modified shortcuts work even while typing inside the editable article,
    // so you can save without leaving edit mode. macOS produces dead keys for Option+E
    // (acute accent) and Option+S (ß), so match on e.code, not e.key.
    if (e.altKey && !e.metaKey && !e.ctrlKey && !e.shiftKey) {
      if (e.code === 'KeyE') {
        if (editBtn) { e.preventDefault(); editBtn.click(); }
        return;
      }
      if (e.code === 'KeyS') {
        e.preventDefault();
        saveMarkdown();
        return;
      }
    }

    // Unmodified shortcuts: skip when typing or when a modifier is held.
    if (e.target.matches('input, textarea, [contenteditable="true"]')) return;
    if (e.metaKey || e.ctrlKey || e.altKey) return;

    if (e.key === '/') {
      e.preventDefault();
      const first = document.querySelector('.toc__link');
      if (first) first.focus();
      return;
    }
    if (e.key === 't' || e.key === 'T') {
      themeBtn.click();
      return;
    }
    if (e.key === 'g' || e.key === 'G') {
      gPrefix = true;
      clearTimeout(gTimer);
      gTimer = setTimeout(() => { gPrefix = false; }, 800);
      return;
    }
    if (gPrefix && /^[1-9]$/.test(e.key)) {
      gPrefix = false;
      clearTimeout(gTimer);
      const topLinks = Array.from(document.querySelectorAll('.toc > .toc__list > .toc__item > .toc__link'));
      const target = topLinks[parseInt(e.key, 10) - 1];
      if (target) target.click();
    }
  });
})();

<script>
(function () {
  const sw = document.getElementById('report-switcher');
  if (!sw) return;
  const input = document.getElementById('report-switcher-input');
  const list = document.getElementById('report-switcher-list');
  const closeBtn = document.getElementById('report-switcher-close');
  const countEl = document.getElementById('report-switcher-count');
  const indexEl = document.getElementById('report-index');
  if (!indexEl) return;

  let data;
  try { data = JSON.parse(indexEl.textContent); } catch (e) { data = { reports: [], current_slug: '' }; }

  let previousFocus = null;
  let activeIndex = 0;
  let currentResults = [];

  function escText(s) {
    return String(s == null ? '' : s).replace(/[&<>]/g, c => ({ '&':'&amp;', '<':'&lt;', '>':'&gt;' }[c]));
  }
  function escAttr(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({ '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;' }[c]));
  }

  function fuzzy(needle, hay) {
    if (!needle) return true;
    let i = 0;
    for (let j = 0; j < hay.length && i < needle.length; j++) {
      if (hay[j] === needle[i]) i++;
    }
    return i === needle.length;
  }

  function filterReports(q) {
    const reports = (data.reports || []).filter(r => r.slug !== data.current_slug);
    if (!q) return reports;
    const needle = q.toLowerCase();
    return reports.filter(r => {
      const hay = (
        (r.title || '') + ' ' +
        ((r.tags || []).join(' ')) + ' ' +
        (r.summary || '') + ' ' +
        (r.eyebrow || '')
      ).toLowerCase();
      return hay.includes(needle) || fuzzy(needle, hay);
    });
  }

  function renderList() {
    const q = input.value.trim();
    const all = filterReports(q);
    const top = all.slice(0, 8);
    currentResults = top;
    activeIndex = Math.min(activeIndex, top.length - 1);
    if (activeIndex < 0) activeIndex = 0;

    if (top.length === 0) {
      list.innerHTML = '<li class="report-switcher__empty">No reports match.</li>';
      countEl.textContent = (data.reports || []).length + ' reports';
      return;
    }

    list.innerHTML = top.map((r, i) => {
      const isActive = i === activeIndex;
      const key = isActive ? '↵' : String(i + 1);
      return '<li role="option" data-href="' + escAttr(r.href) + '" data-index="' + i + '" ' +
        'class="report-switcher__item' + (isActive ? ' is-active' : '') + '" ' +
        'aria-selected="' + (isActive ? 'true' : 'false') + '">' +
        '<span class="report-switcher__item-date">' + escText((r.date || '').slice(5)) + '</span>' +
        '<span class="report-switcher__item-title">' + escText(r.title || '(untitled)') + '</span>' +
        '<span class="report-switcher__item-key">' + escText(key) + '</span>' +
      '</li>';
    }).join('');

    const more = all.length - top.length;
    countEl.textContent = more > 0
      ? (more + ' more · ' + all.length + ' total')
      : (top.length + (top.length === 1 ? ' report' : ' reports'));
  }

  function openSwitcher() {
    if (!sw.hasAttribute('hidden')) return;
    previousFocus = document.activeElement;
    sw.removeAttribute('hidden');
    input.value = '';
    activeIndex = 0;
    renderList();
    requestAnimationFrame(() => input.focus());
    window.dispatchEvent(new CustomEvent('report-switcher-open'));
  }
  function closeSwitcher() {
    if (sw.hasAttribute('hidden')) return;
    sw.setAttribute('hidden', '');
    if (previousFocus && typeof previousFocus.focus === 'function') previousFocus.focus();
    window.dispatchEvent(new CustomEvent('report-switcher-close'));
  }

  function openActive(newTab) {
    const r = currentResults[activeIndex];
    if (!r) return;
    if (newTab) window.open(r.href, '_blank', 'noopener');
    else window.location.href = r.href;
  }

  let renderTimer = 0;
  input.addEventListener('input', () => {
    clearTimeout(renderTimer);
    renderTimer = setTimeout(renderList, 50);
  });

  sw.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') { e.preventDefault(); closeSwitcher(); return; }
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      if (currentResults.length === 0) return;
      activeIndex = (activeIndex + 1) % currentResults.length;
      renderList();
      return;
    }
    if (e.key === 'ArrowUp') {
      e.preventDefault();
      if (currentResults.length === 0) return;
      activeIndex = (activeIndex - 1 + currentResults.length) % currentResults.length;
      renderList();
      return;
    }
    if (e.key === 'Enter') {
      e.preventDefault();
      openActive(e.metaKey || e.ctrlKey);
      return;
    }
    if (/^[1-8]$/.test(e.key) && document.activeElement !== input) {
      const i = parseInt(e.key, 10) - 1;
      if (currentResults[i]) {
        activeIndex = i;
        openActive(false);
      }
      return;
    }
  });

  input.addEventListener('keydown', (e) => {
    if (/^[1-8]$/.test(e.key) && (e.metaKey || e.ctrlKey)) {
      e.preventDefault();
      const i = parseInt(e.key, 10) - 1;
      if (currentResults[i]) {
        activeIndex = i;
        openActive(false);
      }
    }
  });

  list.addEventListener('click', (e) => {
    const li = e.target.closest('.report-switcher__item');
    if (!li) return;
    activeIndex = parseInt(li.getAttribute('data-index'), 10) || 0;
    openActive(e.metaKey || e.ctrlKey);
  });
  closeBtn.addEventListener('click', closeSwitcher);
  sw.addEventListener('mousedown', (e) => {
    if (e.target === sw) closeSwitcher();
  });

  document.addEventListener('keydown', (e) => {
    if ((e.metaKey || e.ctrlKey) && !e.shiftKey && !e.altKey && (e.key === 'o' || e.key === 'O')) {
      const tag = (document.activeElement && document.activeElement.tagName) || '';
      if (tag === 'INPUT' || tag === 'TEXTAREA' || (document.activeElement && document.activeElement.isContentEditable)) {
        return;
      }
      e.preventDefault();
